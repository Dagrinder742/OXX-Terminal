"""
skill.py - Autonomous Cognitive Skill Implementation
"""

import random
import time
from typing import Any, Dict, List, Tuple


class Skill:
    """
    A cognitive agent skill implementing a four-stage decision-making pipeline:
    Perceive -> Reason -> Evaluate -> Act.
    """

    def __init__(self, name: str = "AdaptivePatrolSkill"):
        self.name = name
        # State tracking dictionary
        self.state: Dict[str, Any] = {
            "cycle_count": 0,
            "environment": {},
            "generated_actions": [],
            "chosen_action": None,
            "evaluation_scores": {},
            "execution_log": [],
        }

    def perceive(self) -> Dict[str, Any]:
        """
        Gathers and populates mock environmental metrics into self.state.
        """
        metrics = {
            "battery_level": round(random.uniform(15.0, 100.0), 2),
            "threat_level": round(random.uniform(0.0, 1.0), 2),
            "signal_strength": round(random.uniform(10.0, 95.0), 2),
            "temperature_c": round(random.uniform(18.0, 42.0), 2),
            "ambient_light_lux": round(random.uniform(50.0, 800.0), 1),
        }
        self.state["environment"] = metrics
        return metrics

    def reason(self) -> List[str]:
        """
        Generates 3 contextual action choices based on environmental factors.
        """
        battery = self.state["environment"].get("battery_level", 50.0)
        threat = self.state["environment"].get("threat_level", 0.1)

        # Candidate actions pool
        candidate_pool = [
            f"Enter low-power standby mode (battery at {battery}%)",
            f"Deploy defensive perimeter scan (threat index: {threat})",
            "Transmit diagnostic ping to command relay",
            "Collect sensor telemetry and log to persistent cache",
            "Initiate active pathfinding to next waypoint",
            "Engage high-gain directional transceiver",
        ]

        # Select exactly 3 action choices
        choices = random.sample(candidate_pool, 3)
        self.state["generated_actions"] = choices
        return choices

    def evaluate(self, actions: List[str]) -> Tuple[str, float]:
        """
        Scores actions using a mathematical utility function influenced by
        environmental telemetry and returns the highest-scoring action.
        """
        scores: Dict[str, float] = {}
        battery_weight = self.state["environment"].get("battery_level", 50.0) / 100.0
        threat_weight = self.state["environment"].get("threat_level", 0.5)

        for action in actions:
            # Mathematical scoring formula using random seed and environmental weights
            base_score = random.uniform(0.4, 0.9)
            utility_score = (
                (base_score * 0.5)
                + (battery_weight * 0.25)
                + ((1.0 - threat_weight) * 0.25)
            )
            scores[action] = round(utility_score, 4)

        # Select the choice with the maximum utility score
        best_action = max(scores, key=scores.get)
        best_score = scores[best_action]

        self.state["evaluation_scores"] = scores
        self.state["chosen_action"] = best_action
        return best_action, best_score

    def act(self, action: str, score: float) -> None:
        """
        Executes and prints the chosen action, updating the internal history.
        """
        print(f"  [EXECUTE] Action: '{action}' | Confidence: {score:.4f}")
        self.state["execution_log"].append(
            {
                "cycle": self.state["cycle_count"],
                "action": action,
                "score": score,
            }
        )

    def think(self, iterations: int = 3) -> None:
        """
        Sequentially executes the perceive -> reason -> evaluate -> act pipeline in a loop.
        """
        print(f"=== Starting Skill Loop: {self.name} ({iterations} iterations) ===")

        for i in range(1, iterations + 1):
            self.state["cycle_count"] = i
            print(f"\n--- [CYCLE {i}/{iterations}] ---")

            # 1. Perceive
            print("  [STEP 1: PERCEIVE]")
            env = self.perceive()
            print(f"    Perceived Environment: {env}")

            # 2. Reason
            print("  [STEP 2: REASON]")
            actions = self.reason()
            for idx, action in enumerate(actions, start=1):
                print(f"    Choice {idx}: {action}")

            # 3. Evaluate
            print("  [STEP 3: EVALUATE]")
            chosen_action, score = self.evaluate(actions)
            for action, s in self.state["evaluation_scores"].items():
                print(f"    Score {s:.4f} -> {action}")
            print(f"    Selected Winner: '{chosen_action}' (Score: {score:.4f})")

            # 4. Act
            print("  [STEP 4: ACT]")
            self.act(chosen_action, score)

        print(f"\n=== Skill Loop Completed. Total cycles: {self.state['cycle_count']} ===")


if __name__ == "__main__":
    agent_skill = Skill(name="AutonomousReconSkill")
    agent_skill.think(iterations=3)
