```python
import random

class Skill:
    def __init__(self):
        self.state = {}

    def perceive(self):
        # Mock environmental metrics
        self.state['temperature'] = random.randint(10, 30)
        self.state['humidity'] = random.randint(40, 80)
        self.state['light'] = random.randint(0, 100)

    def reason(self):
        # Generate 3 action choices
        return [
            f"Action 1: Increase temperature by {random.randint(1, 5)} degrees",
            f"Action 2: Decrease humidity by {random.randint(1, 10)}%",
            f"Action 3: Increase light by {random.randint(1, 10)}%"
        ]

    def evaluate(self, actions):
        # Score actions using a random metric
        scores = []
        for action in actions:
            score = random.randint(1, 10)
            scores.append((action, score))
        scores.sort(key=lambda x: x[1], reverse=True)
        return scores[0][0]

    def act(self, action):
        # Print the chosen action
        print(f"Executing: {action}")

    def think(self):
        # Call all four sequentially in a loop
        for _ in range(3):
            self.perceive()
            actions = self.reason()
            chosen_action = self.evaluate(actions)
            self.act(chosen_action)
            print(f"State after action: {self.state}")

# Example usage
skill = Skill()
skill.think()
```
