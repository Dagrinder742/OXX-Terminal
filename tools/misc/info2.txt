import random

class Skill:
    def __init__(self):
        self.state = {}

    def perceive(self):
        # Mock environmental metrics
        self.state['temperature'] = random.uniform(15, 30)
        self.state['humidity'] = random.uniform(40, 80)
        self.state['light'] = random.uniform(0, 100)

    def reason(self):
        # Generate 3 action choices based on state
        actions = [
            f"Move to cooler area ({self.state['temperature']})",
            f"Adjust lights to {self.state['light']}%",
            f"Increase humidity to {self.state['humidity']}%"
        ]
        return actions

    def evaluate(self, actions):
        # Randomly score actions
        scores = {action: random.randint(1, 10) for action in actions}
        return max(scores, key=scores.get)

    def act(self, chosen_action):
        # Print the chosen action
        print(f"Executing: {chosen_action}")

    def think(self):
        # Call all methods sequentially
        self.perceive()
        print("Perceived environment:")
        print(f"Temperature: {self.state['temperature']}")
        print(f"Humidity: {self.state['humidity']}")
        print(f"Light: {self.state['light']}")

        actions = self.reason()
        print("Generated actions:")
        for action in actions:
            print(action)

        chosen_action = self.evaluate(actions)
        print(f"Chosen action: {chosen_action}")

        self.act(chosen_action)

        # Loop to think again
        self.think()
